def lap_to_seconds(lap_time):
    try:
        if pd.isna(lap_time) or lap_time == '-':
            return None
        minutes, seconds = lap_time.split(':')
        return int(minutes) * 60 + float(seconds)
    except:
        return None

df ['fastest_lap (seconds)'] = df['fl_time'].apply(lap_to_seconds)
df[['fastest_lap (seconds)']]
